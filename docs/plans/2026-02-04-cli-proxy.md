# CLI Proxy for AI Agent Tool Validation - Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Extend the Maybe Don't CLI to act as a proxy for CLI commands executed by AI agents, applying the same policy validation (CEL rules, AI validation) used for MCP tool calls.

**Architecture:** The CLI wrapper (`maybe-dont cli -- <command>`) sends validation requests to a REST endpoint (`/api/v1/cli/validate`) on the gateway. The gateway evaluates CEL and AI policies, logs to audit, and returns allow/deny. The CLI then either executes the command via `syscall.Exec` or prints an error.

**Tech Stack:** Go, Cobra CLI, CEL (google/cel-go), HTTP REST, syscall.Exec

**Spec:** See `docs/specs/cli-proxy-for-ai-agents.md` for full requirements.

```go
package gateway

// CLIValidationRequest represents a CLI command validation request.
type CLIValidationRequest struct {
	Command          string         `json:"command"`
	Arguments        []string       `json:"arguments"`
	WorkingDirectory string         `json:"working_directory,omitempty"`
	ClientInfo       *CLIClientInfo `json:"client_info,omitempty"`
}

// CLIClientInfo contains client environment information.
type CLIClientInfo struct {
	Hostname   string `json:"hostname,omitempty"`
	Username   string `json:"username,omitempty"`
	OS         string `json:"os,omitempty"`
	OSVersion  string `json:"os_version,omitempty"`
	Arch       string `json:"arch,omitempty"`
	Shell      string `json:"shell,omitempty"`
	CLIVersion string `json:"cli_version,omitempty"`
}

// CLIValidationResponse represents the validation result.
type CLIValidationResponse struct {
	Allowed            bool              `json:"allowed"`
	ValidationRequired bool              `json:"validation_required"`
	Message            string            `json:"message"`
	ActionReason       string            `json:"action_reason,omitempty"`
	ServerVersion      string            `json:"server_version"`
	ClientVersion      string            `json:"client_version,omitempty"`
	Results            []CLIPolicyResult `json:"results,omitempty"`
}

// CLIPolicyResult represents a single policy evaluation result.
type CLIPolicyResult struct {
	PolicyName string `json:"policy_name"`
	PolicyType string `json:"policy_type"` // "cel" or "ai"
	Action     string `json:"action"`      // "allow", "deny", "redact"
	Message    string `json:"message,omitempty"`
}

// CLIValidationError represents an error response.
type CLIValidationError struct {
	Error   string `json:"error"`
	Message string `json:"message"`
}
```

```go
package gateway

import (
	"encoding/json"
	"net/http"
	"slices"

	"github.com/maybedont/maybe-dont/internal/config"
)

// CLIValidationHandlerConfig configures the CLI validation HTTP handler.
type CLIValidationHandlerConfig struct {
	Enabled          bool
	ValidateCommands []string
	Logger           *config.SessionLogger
	Version          string
	// Future: CELEngine, AIEngine for policy evaluation
}

// CLIValidationHandler handles /api/v1/cli/validate requests.
type CLIValidationHandler struct {
	config CLIValidationHandlerConfig
}

// NewCLIValidationHandler creates a new CLI validation handler.
func NewCLIValidationHandler(cfg CLIValidationHandlerConfig) *CLIValidationHandler {
	return &CLIValidationHandler{config: cfg}
}

// ServeHTTP handles CLI validation requests.
func (h *CLIValidationHandler) ServeHTTP(w http.ResponseWriter, r *http.Request) {
	w.Header().Set("Content-Type", "application/json")

	// Check if CLI validation is enabled
	if !h.config.Enabled {
		h.writeError(w, http.StatusBadRequest, "cli_validation_disabled",
			"CLI validation is not enabled on this gateway. Set cli_request_validation.enabled: true in configuration.")
		return
	}

	// Validate Content-Type
	if r.Header.Get("Content-Type") != "application/json" {
		h.writeError(w, http.StatusBadRequest, "invalid_content_type",
			"Content-Type must be application/json")
		return
	}

	// Parse request body
	var req CLIValidationRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		h.writeError(w, http.StatusBadRequest, "invalid_request",
			"Failed to parse request body: "+err.Error())
		return
	}

	// Validate required fields
	if req.Command == "" {
		h.writeError(w, http.StatusBadRequest, "missing_command",
			"Required field 'command' is empty")
		return
	}

	// Check if command requires validation
	if !h.requiresValidation(req.Command) {
		resp := CLIValidationResponse{
			Allowed:            true,
			ValidationRequired: false,
			Message:            "Command does not require validation",
			ServerVersion:      h.config.Version,
			ClientVersion:      h.getClientVersion(&req),
		}
		json.NewEncoder(w).Encode(resp)
		return
	}

	// Command requires validation - evaluate policies
	// TODO: Integrate with CEL and AI engines in Task 2.4
	resp := CLIValidationResponse{
		Allowed:            true, // Default allow when no policies configured
		ValidationRequired: true,
		Message:            "Command approved by policy",
		ServerVersion:      h.config.Version,
		ClientVersion:      h.getClientVersion(&req),
		Results:            []CLIPolicyResult{},
	}
	json.NewEncoder(w).Encode(resp)
}

// requiresValidation checks if the command is in the validate_commands list.
func (h *CLIValidationHandler) requiresValidation(command string) bool {
	// "*" matches all commands
	if slices.Contains(h.config.ValidateCommands, "*") {
		return true
	}
	return slices.Contains(h.config.ValidateCommands, command)
}

// getClientVersion extracts client version from request.
func (h *CLIValidationHandler) getClientVersion(req *CLIValidationRequest) string {
	if req.ClientInfo != nil {
		return req.ClientInfo.CLIVersion
	}
	return ""
}

// writeError writes a JSON error response.
func (h *CLIValidationHandler) writeError(w http.ResponseWriter, status int, code, message string) {
	w.WriteHeader(status)
	json.NewEncoder(w).Encode(CLIValidationError{
		Error:   code,
		Message: message,
	})
}
```

```go
package cliproxy

import (
	"bytes"
	"context"
	"encoding/json"
	"fmt"
	"net/http"
	"time"
)

// ValidationRequest is the request sent to the gateway.
type ValidationRequest struct {
	Command          string      `json:"command"`
	Arguments        []string    `json:"arguments"`
	WorkingDirectory string      `json:"working_directory,omitempty"`
	ClientInfo       *ClientInfo `json:"client_info,omitempty"`
}

// ClientInfo contains client environment information.
type ClientInfo struct {
	Hostname   string `json:"hostname,omitempty"`
	Username   string `json:"username,omitempty"`
	OS         string `json:"os,omitempty"`
	OSVersion  string `json:"os_version,omitempty"`
	Arch       string `json:"arch,omitempty"`
	Shell      string `json:"shell,omitempty"`
	CLIVersion string `json:"cli_version,omitempty"`
}

// ValidationResponse is the response from the gateway.
type ValidationResponse struct {
	Allowed            bool           `json:"allowed"`
	ValidationRequired bool           `json:"validation_required"`
	Message            string         `json:"message"`
	ActionReason       string         `json:"action_reason,omitempty"`
	ServerVersion      string         `json:"server_version"`
	ClientVersion      string         `json:"client_version,omitempty"`
	Results            []PolicyResult `json:"results,omitempty"`
}

// PolicyResult represents a single policy evaluation result.
type PolicyResult struct {
	PolicyName string `json:"policy_name"`
	PolicyType string `json:"policy_type"`
	Action     string `json:"action"`
	Message    string `json:"message,omitempty"`
}

// ErrorResponse represents an error from the gateway.
type ErrorResponse struct {
	Error   string `json:"error"`
	Message string `json:"message"`
}

// ClientConfig configures the validation client.
type ClientConfig struct {
	ServerURL string
	Timeout   time.Duration
	ClientID  string // Optional client ID for audit attribution
}

// Client communicates with the Maybe Don't gateway.
type Client struct {
	config     ClientConfig
	httpClient *http.Client
}

// NewClient creates a new validation client.
func NewClient(cfg ClientConfig) *Client {
	timeout := cfg.Timeout
	if timeout == 0 {
		timeout = 30 * time.Second
	}

	return &Client{
		config: cfg,
		httpClient: &http.Client{
			Timeout: timeout,
		},
	}
}

// Validate sends a validation request to the gateway.
func (c *Client) Validate(ctx context.Context, req ValidationRequest) (*ValidationResponse, error) {
	body, err := json.Marshal(req)
	if err != nil {
		return nil, fmt.Errorf("failed to marshal request: %w", err)
	}

	url := c.config.ServerURL + "/api/v1/cli/validate"
	httpReq, err := http.NewRequestWithContext(ctx, http.MethodPost, url, bytes.NewReader(body))
	if err != nil {
		return nil, fmt.Errorf("failed to create request: %w", err)
	}

	httpReq.Header.Set("Content-Type", "application/json")
	if c.config.ClientID != "" {
		httpReq.Header.Set("X-Maybe-Dont-Client-ID", c.config.ClientID)
	}

	resp, err := c.httpClient.Do(httpReq)
	if err != nil {
		return nil, fmt.Errorf("failed to send request: %w", err)
	}
	defer resp.Body.Close()

	if resp.StatusCode >= 400 {
		var errResp ErrorResponse
		if err := json.NewDecoder(resp.Body).Decode(&errResp); err != nil {
			return nil, fmt.Errorf("server returned status %d", resp.StatusCode)
		}
		return nil, fmt.Errorf("%s: %s", errResp.Error, errResp.Message)
	}

	var valResp ValidationResponse
	if err := json.NewDecoder(resp.Body).Decode(&valResp); err != nil {
		return nil, fmt.Errorf("failed to decode response: %w", err)
	}

	return &valResp, nil
}
```

```go
package cmd

import (
	"context"
	"fmt"
	"os"
	"time"

	"github.com/maybedont/maybe-dont/internal/cliproxy"
	"github.com/spf13/cobra"
)

var (
	cliServer  string
	cliTimeout time.Duration
	cliDryRun  bool
)

var cliCmd = &cobra.Command{
	Use:   "cli [flags] -- <command> [args...]",
	Short: "Validate and execute CLI commands through the gateway",
	Long: `Routes CLI commands through the Maybe Don't gateway for validation.

The -- separator is REQUIRED to separate cli flags from the command to execute.

Examples:
  maybe-dont cli -- gh pr comment 123 --body "LGTM"
  maybe-dont cli --server https://gateway:8443 -- aws s3 ls
  maybe-dont cli --dry-run -- kubectl delete pod my-pod
`,
	RunE:              runCLI,
	DisableFlagParsing: false,
}

func init() {
	rootCmd.AddCommand(cliCmd)

	cliCmd.Flags().StringVarP(&cliServer, "server", "s", "http://localhost:8080", "Gateway base URL")
	cliCmd.Flags().DurationVar(&cliTimeout, "timeout", 30*time.Second, "Validation request timeout")
	cliCmd.Flags().BoolVar(&cliDryRun, "dry-run", false, "Validate only, don't execute")
}

func runCLI(cmd *cobra.Command, args []string) error {
	// Find -- separator
	separatorIdx := -1
	for i, arg := range os.Args {
		if arg == "--" {
			separatorIdx = i
			break
		}
	}

	if separatorIdx == -1 {
		return fmt.Errorf("missing command separator\nUsage: maybe-dont cli [flags] -- <command> [args...]")
	}

	// Extract command and args after --
	commandArgs := os.Args[separatorIdx+1:]
	if len(commandArgs) == 0 {
		return fmt.Errorf("no command specified\nUsage: maybe-dont cli [flags] -- <command> [args...]")
	}

	command := commandArgs[0]
	cmdArgs := commandArgs[1:]

	// Get client ID from environment
	clientID := os.Getenv("MAYBE_DONT_CLIENT_ID")

	// Create client
	client := cliproxy.NewClient(cliproxy.ClientConfig{
		ServerURL: cliServer,
		Timeout:   cliTimeout,
		ClientID:  clientID,
	})

	// Collect client info
	clientInfo := cliproxy.CollectClientInfo()

	// Get working directory
	wd, _ := os.Getwd()

	// Validate command
	ctx, cancel := context.WithTimeout(context.Background(), cliTimeout)
	defer cancel()

	resp, err := client.Validate(ctx, cliproxy.ValidationRequest{
		Command:          command,
		Arguments:        cmdArgs,
		WorkingDirectory: wd,
		ClientInfo:       clientInfo,
	})

	if err != nil {
		// Fail-open: allow command if gateway unreachable
		fmt.Fprintf(os.Stderr, "Warning: Unable to reach validation gateway, proceeding without validation\n")
		if cliDryRun {
			fmt.Println("Dry run: would execute command (gateway unreachable)")
			return nil
		}
		return cliproxy.Execute(command, cmdArgs)
	}

	if !resp.Allowed {
		fmt.Fprintf(os.Stderr, "Error: Command blocked by security policy\n")
		if resp.ActionReason != "" {
			fmt.Fprintf(os.Stderr, "Reason: %s\n", resp.Message)
		}
		os.Exit(1)
	}

	if cliDryRun {
		fmt.Println("Dry run: command would be allowed")
		if resp.ValidationRequired {
			fmt.Println("Validation required: yes")
		} else {
			fmt.Println("Validation required: no (command not in allowlist)")
		}
		return nil
	}

	// Execute command
	return cliproxy.Execute(command, cmdArgs)
}
```
